cmake_minimum_required(VERSION 3.20)
project(dt_render_native C CXX)
set(AOM_TARGET_CPU generic CACHE STRING "No assembler dependency" FORCE)
set(ENABLE_APPS OFF CACHE BOOL "" FORCE)
set(ENABLE_DOCS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(CONFIG_AV1_DECODER OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/aom" aom-build EXCLUDE_FROM_ALL)
target_include_directories(aom INTERFACE "$<BUILD_INTERFACE:${SOURCES}/aom>" "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/aom-build>")
set(AVIF_CODEC_AOM SYSTEM CACHE STRING "" FORCE)
set(AVIF_CODEC_AOM_DECODE OFF CACHE BOOL "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_ENABLE_WERROR OFF CACHE BOOL "" FORCE)
set(AOM_LIBRARY aom CACHE STRING "" FORCE)
set(AOM_INCLUDE_DIR "${SOURCES}/aom" CACHE PATH "" FORCE)
add_subdirectory("${SOURCES}/avif" avif-build EXCLUDE_FROM_ALL)
add_library(dt_render_codec STATIC codec.c)
target_include_directories(dt_render_codec PRIVATE "${SOURCES}/avif/include")
target_link_libraries(dt_render_codec PRIVATE avif aom)
install(TARGETS dt_render_codec avif aom ARCHIVE DESTINATION lib)
